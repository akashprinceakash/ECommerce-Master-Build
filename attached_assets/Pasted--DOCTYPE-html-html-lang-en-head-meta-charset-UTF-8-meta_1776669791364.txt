function createBodyGeometry() {
  const shape = new THREE.Shape();
  shape.moveTo(-0.52, -0.88);
  shape.lineTo(-0.52, 0.55);
  shape.bezierCurveTo(-0.52, 0.7, -0.38, 0.78, -0.28, 0.78);
  shape.lineTo(0.28, 0.78);
  shape.bezierCurveTo(0.38, 0.78, 0.52, 0.7, 0.52, 0.55);
  shape.lineTo(0.52, -0.88);
  shape.lineTo(-0.52, -0.88);

  const geo = new THREE.ShapeGeometry(shape);
  geo.computeVertexNormals();

  // Give it thickness
  const extruded = new THREE.ExtrudeGeometry(shape, { depth: 0.05, bevelEnabled: false });
  return extruded;
}

function mergeGeos(geos) {
  if (geos.length === 1) return geos[0];
  const { BufferGeometry, BufferAttribute } = THREE;
  const merged = new BufferGeometry();
  let totalVerts = 0;
  for (const g of geos) totalVerts += g.attributes.position.count;
  const positions = new Float32Array(totalVerts * 3);
  const normals = new Float32Array(totalVerts * 3);
  let offset = 0;
  for (const g of geos) {
    positions.set(g.attributes.position.array, offset * 3);
    if (g.attributes.normal) normals.set(g.attributes.normal.array, offset * 3);
    offset += g.attributes.position.count;
  }
  merged.setAttribute('position', new BufferAttribute(positions, 3));
  merged.setAttribute('normal', new BufferAttribute(normals, 3));
  return merged;
}

  ctx.textBaseline = 'middle';
  ctx.fillText(txt, 256, 64);

  const texture = new THREE.CanvasTexture(canvas);
  const mat = new THREE.SpriteMaterial({ map: texture, transparent: true });
  textSprite = new THREE.Sprite(mat);
  textSprite.scale.set(0.6, 0.15, 1);
  textSprite.position.copy(getPlacementPos(state.textPos));
  shirtGroup.add(textSprite);
  updateSummary();
}

function removeText() {
  if (textSprite) { shirtGroup.remove(textSprite); textSprite = null; }
  document.getElementById('text-content').value = '';
  state.text = '';
  updateSummary();
}

/* ─── Logo overlay ─── */
function handleLogoUpload(input) {
  const file = input.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = e => {
    state.logoUrl = e.target.result;
    const img = document.getElementById('logo-preview');
    img.src = e.target.result;
    document.getElementById('logo-preview-wrap').style.display = 'block';
    updateSummary();
    showToast('Logo uploaded! Click "Apply Logo" to place it.');
  };
  reader.readAsDataURL(file);
}

function applyLogo() {
  if (!state.logoUrl) { showToast('Please upload a logo first.'); return; }
  if (logoSprite) { shirtGroup.remove(logoSprite); logoSprite = null; }

  const img = new Image();
  img.onload = () => {
    const canvas = document.createElement('canvas');
    canvas.width = 256; canvas.height = 256;
    const ctx = canvas.getContext('2d');
    ctx.clearRect(0, 0, 256, 256);
    ctx.drawImage(img, 0, 0, 256, 256);
    const tex = new THREE.CanvasTexture(canvas);
    const mat = new THREE.SpriteMaterial({ map: tex, transparent: true });
    logoSprite = new THREE.Sprite(mat);
    const sz = state.logoSize / 256;
    logoSprite.scale.set(sz, sz, 1);
    logoSprite.position.copy(getPlacementPos(state.logoPos));
    shirtGroup.add(logoSprite);
    updateSummary();
  };
  img.src = state.logoUrl;
}

function removeLogo() {
  if (logoSprite) { shirtGroup.remove(logoSprite); logoSprite = null; }
  state.logoUrl = null;
  document.getElementById('logo-preview-wrap').style.display = 'none';
  updateSummary();
}

function updateLogoSize(val) {
  state.logoSize = parseInt(val);
  document.getElementById('logo-size-val').textContent = val + 'px';
  if (logoSprite) {
    const sz = state.logoSize / 256;
    logoSprite.scale.set(sz, sz, 1);
  }
}

function getPlacementPos(pos) {
  const positions = {
    'front-chest': new THREE.Vector3(-0.16, 0.55, 0.08),
    'front-center': new THREE.Vector3(0, 0.1, 0.08),
    'back-top': new THREE.Vector3(0, 0.55, -0.08),
    'back-center': new THREE.Vector3(0, 0.1, -0.08),
    'sleeve-left': new THREE.Vector3(-0.72, 0.58, 0.05),
    'sleeve-right': new THREE.Vector3(0.72, 0.58, 0.05),
  };
  return positions[pos] || positions['front-chest'];
}

/* ─── Animate ─── */
function animate() {
  requestAnimationFrame(animate);
  currentY += (targetY - currentY) * 0.08;
  shirtGroup.rotation.y = currentY;
  if (state.autoRotate) {
    targetY += 0.008;
  }
  renderer.render(scene, camera);
}

/* ─── Drag to rotate ─── */
function setupDrag() {
  const el = renderer.domElement;
  el.addEventListener('mousedown', e => { isDragging = true; prevMouse = {x:e.clientX, y:e.clientY}; });
  el.addEventListener('mousemove', e => {
    if (!isDragging) return;
    const dx = e.clientX - prevMouse.x;
    targetY += dx * 0.01;
    prevMouse = {x:e.clientX, y:e.clientY};
  });
  el.addEventListener('mouseup', () => isDragging = false);
  el.addEventListener('mouseleave', () => isDragging = false);

  // Touch
  el.addEventListener('touchstart', e => { isDragging = true; prevMouse = {x:e.touches[0].clientX, y:e.touches[0].clientY}; });
  el.addEventListener('touchmove', e => {
    if (!isDragging) return;
    const dx = e.touches[0].clientX - prevMouse.x;
    targetY += dx * 0.01;
    prevMouse = {x:e.touches[0].clientX, y:e.touches[0].clientY};
  });
  el.addEventListener('touchend', () => isDragging = false);
}

/* ─── Camera views ─── */
function setView(view, btn) {
  document.querySelectorAll('.view-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  state.autoRotate = false;
  if (view === 'front') { targetY = 0; }
  else if (view === 'back') { targetY = Math.PI; }
  else if (view === 'side') { targetY = Math.PI / 2; }
  else if (view === 'rotate') { state.autoRotate = true; }
}

  });
}
