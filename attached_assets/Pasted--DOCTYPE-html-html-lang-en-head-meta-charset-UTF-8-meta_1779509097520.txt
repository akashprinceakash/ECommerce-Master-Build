//  GT SWATCH DRAWING (canvas-based preview)

function drawGtSwatch(canvas, style, primary, accent, tertiary) {
  const ctx = canvas.getContext('2d');
  const W = canvas.width, H = canvas.height;
  ctx.clearRect(0,0,W,H);
  const p = primary || style.primary;
  const a = accent  || style.accent;
  const t = tertiary|| style.tertiary || null;
  switch(style.group) {
    case 'classic':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a;
      ctx.fillRect(0,0,W,Math.round(H*0.16));
      ctx.fillRect(0,H-Math.round(H*0.12),W,Math.round(H*0.12));
      ctx.fillRect(0,H-Math.round(H*0.22),W,Math.round(H*0.06));
      break;
    case 'sport-side':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a;
      ctx.fillRect(0,0,Math.round(W*0.22),H);
      ctx.fillRect(W-Math.round(W*0.22),0,Math.round(W*0.22),H);
      if(t){ctx.fillStyle=t;ctx.fillRect(0,0,W,Math.round(H*0.12));}
      break;
    case 'triple':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a; ctx.fillRect(0,0,Math.round(W*0.22),H); ctx.fillRect(W-Math.round(W*0.22),0,Math.round(W*0.22),H);
      if(t){ctx.fillStyle=t;ctx.fillRect(0,0,W,Math.round(H*0.14));}
      break;
    case 'wave':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a;
      ctx.beginPath();
      ctx.moveTo(0,H*0.6); ctx.quadraticCurveTo(W/2,H*0.35,W,H*0.6);
      ctx.lineTo(W,H); ctx.lineTo(0,H); ctx.closePath(); ctx.fill();
      break;
    case 'hourglass':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a;
      ctx.beginPath();
      ctx.moveTo(W*0.22,0); ctx.lineTo(W*0.78,0);
      ctx.lineTo(W*0.62,H*0.55); ctx.lineTo(W*0.78,H);
      ctx.lineTo(W*0.22,H); ctx.lineTo(W*0.38,H*0.55);
      ctx.closePath(); ctx.fill();
      break;
    case 'pinstripe':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.strokeStyle = a; ctx.lineWidth = 1.5;
      for(let x=8;x<W;x+=Math.round(W/6)){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();}
      break;
    case 'raglan':
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
      ctx.fillStyle = a;
      ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(W*0.45,0);ctx.lineTo(W*0.18,H*0.4);ctx.lineTo(0,H*0.22);ctx.closePath();ctx.fill();
      ctx.beginPath();ctx.moveTo(W,0);ctx.lineTo(W*0.55,0);ctx.lineTo(W*0.82,H*0.4);ctx.lineTo(W,H*0.22);ctx.closePath();ctx.fill();
      ctx.fillRect(0,0,W,Math.round(H*0.1));
      ctx.fillRect(0,H-Math.round(H*0.12),W,Math.round(H*0.12));
      break;
    default:
      ctx.fillStyle = p; ctx.fillRect(0,0,W,H);
  }
}

let _syncTimer = null;
function scheduleSync() { clearTimeout(_syncTimer); _syncTimer = setTimeout(syncTexture, 120); }

async function syncTexture() {
  const mv = S.mv, fc = S.fc;
  if(!mv||!fc) return;
  try {
    fc.renderAll();
    await new Promise(r => requestAnimationFrame(r));
    fc.renderAll();
    await new Promise(r => requestAnimationFrame(r));
    const rawEl = fc.getElement ? fc.getElement() : (fc.lowerCanvasEl);
    const dataUrl = rawEl ? rawEl.toDataURL('image/png',1.0) : fc.toDataURL({format:'png',quality:1,multiplier:1});
    if(!dataUrl||dataUrl.length<100) return;
    const tex = await mv.createTexture(dataUrl);
    const model = mv.model;
    if(!model?.materials?.length) return;
    for(const mat of model.materials){
      const pbr = mat?.pbrMetallicRoughness; if(!pbr) continue;
      const slot = pbr.baseColorTexture; if(!slot) continue;
      try { slot.setTexture(tex); pbr.setBaseColorFactor([1,1,1,1]); break; }
      catch { try { if(slot&&typeof slot.texture!=='undefined'){slot.texture=tex;pbr.setBaseColorFactor([1,1,1,1]);} break; } catch {} }
    }
  } catch(e) { console.warn('[KaSha] syncTexture:', e); }
}

function applyZoneColor(zone, hex) {
  if(!S.fc) return;
  // Remove existing zone color rect
  S.fc.getObjects().filter(o => o.data && o.data.kashaZoneColor === zone).forEach(o => S.fc.remove(o));
  if(!hex) { S.zoneColors[zone] = ''; renderZoneList(); S.fc.renderAll(); scheduleSync(); return; }
  const p = ZONE_PRESETS[zone];
  const rect = new fabric.Rect({ left:p.left, top:p.top, width:p.w, height:p.h, fill:hex, selectable:false, evented:false, originX:'left', originY:'top' });
  rect.data = { kashaZoneColor: zone };
  S.fc.add(rect);
  // send zone rects to back, keep GT base behind them
  S.fc.sendToBack(rect);
  const gtBase = S.fc.getObjects().find(o => o.data && o.data.tag === '__gtBase__');
  if(gtBase) S.fc.sendToBack(gtBase);
  S.zoneColors[zone] = hex;
  renderZoneList();
  S.fc.renderAll();
  scheduleSync();
  updateSummary();
}

function handleZoneColorChange(zone, hex) {
  if(S.fullBody) { ALL_ZONES.forEach(z => applyZoneColor(z, hex)); }
  else { applyZoneColor(zone, hex); }
}

// ═══════════════════════════════════════════════════════════════
//  GT STYLE SYSTEM (draws zone rects on Fabric canvas)
// ═══════════════════════════════════════════════════════════════

function applyGtStyle(styleId, primary, accent, tertiary) {
  if(!S.fc) return;
  const style = GT_STYLES.find(s => s.id === styleId); if(!style) return;
  primary   = primary   || style.primary;
  accent    = accent    || style.accent;
  tertiary  = tertiary  || style.tertiary || null;

  // Clear existing GT base and zone color rects
  S.fc.getObjects().filter(o => o.data && (o.data.tag === '__gtBase__' || o.data.kashaGt)).forEach(o => S.fc.remove(o));

  // Helper: draw a colored rectangle on the canvas
  const addRect = (z, color, selectable=false) => {
    const p = ZONE_PRESETS[z];
    const r = new fabric.Rect({ left:p.left, top:p.top, width:p.w, height:p.h, fill:color, selectable:false, evented:false, originX:'left', originY:'top' });
    r.data = { tag:'__gtBase__', kashaGt:styleId };
    return r;
  };

  let rects = [];
  switch(style.group) {
    case 'classic':
      rects = ALL_ZONES.map(z => addRect(z, ['collar','leftSleeve','rightSleeve'].includes(z) ? accent : primary));
      break;
    case 'sport-side':
      rects = ALL_ZONES.map(z => addRect(z, ['leftSleeve','rightSleeve','collar'].includes(z) ? accent : primary));
      break;
    case 'triple':
      rects = ALL_ZONES.map(z => {
        if(['leftSleeve','rightSleeve'].includes(z)) return addRect(z, accent);
        if(z === 'collar') return addRect(z, tertiary||accent);
        return addRect(z, primary);
      });
      break;
    case 'wave':
    case 'hourglass':
      rects = ALL_ZONES.map(z => addRect(z, ['collar','leftSleeve','rightSleeve'].includes(z) ? accent : primary));
      break;
    case 'pinstripe':
      rects = ALL_ZONES.map(z => addRect(z, primary));
      // Add stripe lines on front zone
      const fp = ZONE_PRESETS.front;
      const stripCount = 6;
      for(let i=1;i<stripCount;i++){
        const x = fp.left + Math.round(fp.w * i/stripCount);
        const line = new fabric.Line([x, fp.top, x, fp.top+fp.h], { stroke:accent, strokeWidth:3, selectable:false, evented:false });
        line.data = { tag:'__gtBase__', kashaGt:styleId };
        rects.push(line);
      }
      const bp = ZONE_PRESETS.back;
      for(let i=1;i<stripCount;i++){
        const x = bp.left + Math.round(bp.w * i/stripCount);
        const line = new fabric.Line([x, bp.top, x, bp.top+bp.h], { stroke:accent, strokeWidth:3, selectable:false, evented:false });
        line.data = { tag:'__gtBase__', kashaGt:styleId };
        rects.push(line);
      }
      break;
    case 'raglan':
      rects = ALL_ZONES.map(z => addRect(z, ['leftSleeve','rightSleeve'].includes(z) ? accent : primary));
      // Collar also accent
      const collarRects = rects.filter(r => r.data && r.data.tag === '__gtBase__');
      break;
    default:
      rects = ALL_ZONES.map(z => addRect(z, primary));
  }

  rects.forEach(r => { S.fc.add(r); S.fc.sendToBack(r); });

  // Set canvas background to primary
  S.fc.backgroundColor = primary;
  S.activeGtStyleId = styleId;
  S.gtColors = { primary, accent, tertiary };
  S.canvasBg = primary;

  // Sync the canvas bg color input
  const bgInp = document.getElementById('canvas-bg-color');
  if(bgInp) bgInp.value = primary;

  S.fc.renderAll();
  scheduleSync();
  updateGtActivePanel();
  updateSummary();
}

function updateGtStyle() {
  if(!S.activeGtStyleId) return;
  applyGtStyle(S.activeGtStyleId, S.gtColors.primary, S.gtColors.accent, S.gtColors.tertiary);
}

function applyAllOverPrint() {
  const p = PATTERNS.find(x => x.id === S.activePrintId); if(!p||!S.fc) return;
  const img = new Image(); img.crossOrigin='anonymous';
  img.onload = () => {
    const off = document.createElement('canvas'); off.width=off.height=384;
    const ctx = off.getContext('2d');
    ctx.imageSmoothingEnabled=true; ctx.imageSmoothingQuality='high';
    ctx.drawImage(img,0,0,384,384);
    const pat = new fabric.Pattern({ source:off, repeat:'repeat' });
    S.fc.backgroundColor = pat;
    S.fc.renderAll();
    S.allOverPrintId = p.id;
    scheduleSync();
    renderPrintGrid();
    document.getElementById('clear-all-over-btn').style.display='block';
    updateSummary();
  };
  img.onerror = () => {
    // Fallback: draw CSS gradient pattern on canvas background
    const off = document.createElement('canvas'); off.width=off.height=256;
    const ctx = off.getContext('2d');
    const grd = ctx.createRadialGradient(128,128,10,128,128,128);
    const [c1,c2] = p.css.includes('#') ? p.css.match(/#[0-9a-fA-F]{3,6}/g) : ['#7a2030','#f5c84a'];
    grd.addColorStop(0, c1); grd.addColorStop(1, c2);
    ctx.fillStyle = grd; ctx.fillRect(0,0,256,256);
    const pat = new fabric.Pattern({ source:off, repeat:'repeat' });
    S.fc.backgroundColor = pat;
    S.fc.renderAll();
    S.allOverPrintId = p.id;
    scheduleSync();
    renderPrintGrid();
    updateSummary();
  };
  img.src = p.file;
}

  const items = window._kashaCart;
  alert(`Cart (${items.length} item${items.length>1?'s':''}):\n\n`+items.map((i,n)=>`${n+1}. ${i.product||'Design'} — ${i.size} × ${i.qty}`).join('\n'));

//  SKU COMPUTATION

function getSku() {
  const p = S.product;
  if(!p) return '—';
  const bodyCode = (p.name||'').replace(/\s+/g,'').toUpperCase().slice(0,8);
  const sizeCode = S.size==='Custom'?'C':S.size;
  let designCode = '';
  if(S.activeGtStyleId) designCode = S.activeGtStyleId;
  else if(S.activePrintId) {
    const idx = PATTERNS.findIndex(x=>x.id===S.activePrintId);
    designCode = idx>=0 ? `GP${String(idx+1).padStart(3,'0')}` : '';
  } else { designCode = sizeCode; }
  return [bodyCode, designCode, sizeCode].filter(Boolean).join(' ');
}

    const styles = GT_STYLES.filter(s => s.group === group.id);

        <span>${group.label} <span style="opacity:.5;font-weight:400">(${styles.length})</span></span>
        <span style="font-size:9px">${isOpen?'▲':'▼'}</span>
      </button>
      ${isOpen?`<div class="gt-swatches">${styles.map(st=>{

        const swId = 'gts_'+st.id;

const TAB_CONFIG = {
  printed: ['text','logo','prints','colors','canvas'],
  pattern: ['text','logo','colors','canvas'],
  solid:   ['text','logo','colors','prints','canvas'],
};

const TAB_LABELS = { text:'Text', logo:'Logo', prints:'Prints', colors:'Colors', canvas:'Canvas' };

function renderTabs(type) {
  const tabs = TAB_CONFIG[type]||TAB_CONFIG.printed;
  const bar = document.getElementById('tab-bar'); if(!bar) return;
  bar.innerHTML = tabs.map(t=>`<button class="tab${t==='text'?' active':''}" id="tab-btn-${t}" onclick="activateTab('${t}')">${TAB_LABELS[t]}</button>`).join('');
  tabs.forEach(t => {
    const c = document.getElementById('tab-'+t);
    if(c) c.classList.toggle('active', t==='text');
  });
  // For pattern type, hide GT picker from colors (show locked style)
  if(type==='pattern') {
    const gts = document.getElementById('gt-styles-section');
    if(gts) gts.style.display = 'none';
    // Show locked pattern notice
    const acPnl = document.getElementById('gt-active-panel');
    if(acPnl && S.activeGtStyleId) updateGtActivePanel();
  }
  if(type==='solid') {
    const gts = document.getElementById('gt-styles-section');
    if(gts) gts.style.display = 'none'; // hide GT for solid, just palette
  }
}

function activateTab(t) {
  const tabs = TAB_CONFIG[S.productType]||TAB_CONFIG.printed;
  tabs.forEach(tab => {
    const btn = document.getElementById('tab-btn-'+tab);
    const content = document.getElementById('tab-'+tab);
    if(btn) btn.classList.toggle('active', tab===t);
    if(content) content.classList.toggle('active', tab===t);
  });
  if(t==='canvas') { setTimeout(scaleFabricPreview, 50); }
}

function buildThumb(product) {
  // SVG T-shirt silhouette with product-appropriate styling
  const type = product.type;
  const colors = product.colors||['#C5D3DE','#ffffff'];
  const c1=colors[0]||'#C5D3DE', c2=colors[1]||'#362223';

  if(type==='pattern' && product.gtStyle) {
    const st = GT_STYLES.find(s=>s.id===product.gtStyle)||{group:'classic',primary:c1,accent:c2};
    const svgId = 'thumb_'+product.id;
    // We'll use a canvas for GT swatch rendering
    return `<canvas id="${svgId}" width="300" height="400" style="width:100%;height:100%" 
      data-gt="${product.gtStyle}" data-primary="${st.primary}" data-accent="${st.accent}" data-group="${st.group}"></canvas>`;
  }

  // Printed or Solid — SVG with appropriate fill
  const fill = type==='printed' ? `url(#p${product.id})` : c1;
  const accentFill = type==='solid' ? c2||c1 : c2||'rgba(255,255,255,0.3)';

  return `<svg viewBox="0 0 300 400" xmlns="http://www.w3.org/2000/svg">
    <defs>
      ${type==='printed'?`<radialGradient id="p${product.id}" cx="50%" cy="50%" r="70%">
        <stop offset="0%" stop-color="${c1}"/>
        <stop offset="60%" stop-color="${c2||'#888'}"/>
        ${colors[2]?`<stop offset="100%" stop-color="${colors[2]}"/>`:''}
      </radialGradient>`:''}
    </defs>
    <!-- T-shirt body -->
    <path d="M80,70 L40,110 L70,125 L70,330 L230,330 L230,125 L260,110 L220,70 L190,45 Q150,30 110,45 Z" fill="${fill}" stroke="rgba(255,255,255,0.08)" stroke-width="1"/>
    <!-- collar -->
    <path d="M110,45 Q150,65 190,45 Q175,80 150,85 Q125,80 110,45 Z" fill="${accentFill}" opacity="0.9"/>
    <!-- left sleeve -->
    <path d="M80,70 L40,110 L70,125 L85,85 Z" fill="${accentFill}" opacity="0.85"/>
    <!-- right sleeve -->
    <path d="M220,70 L260,110 L230,125 L215,85 Z" fill="${accentFill}" opacity="0.85"/>
    ${type==='printed'?`<text x="150" y="220" text-anchor="middle" font-family="serif" font-size="11" fill="rgba(255,255,255,0.3)" letter-spacing="2">CUSTOM PRINT</text>`:''}
    ${type==='solid'?`<text x="150" y="220" text-anchor="middle" font-family="serif" font-size="18" font-weight="bold" fill="rgba(255,255,255,0.12)" letter-spacing="4">KaSha</text>`:''}
  </svg>`;
}

function drawPatternThumbs() {
  // Render GT pattern thumbs using canvas after DOM update
  requestAnimationFrame(() => {
    PRODUCTS.filter(p=>p.type==='pattern'&&p.gtStyle).forEach(p => {
      const c = document.getElementById('thumb_'+p.id);
      if(!c) return;
      const st = GT_STYLES.find(s=>s.id===p.gtStyle); if(!st) return;
      const ctx = c.getContext('2d');
      const W=c.width,H=c.height;
      // Draw t-shirt silhouette shape
      ctx.clearRect(0,0,W,H);
      // Background gradient
      const bg = ctx.createLinearGradient(0,0,0,H);
      bg.addColorStop(0,'#1a1612'); bg.addColorStop(1,'#0c0a08');
      ctx.fillStyle=bg; ctx.fillRect(0,0,W,H);
      // Draw shirt shape with GT colors
      const bodyPath = new Path2D('M80,70 L40,110 L70,125 L70,330 L230,330 L230,125 L260,110 L220,70 L190,45 Q150,30 110,45 Z');
      ctx.fillStyle = st.primary;
      ctx.fill(bodyPath);
      ctx.strokeStyle='rgba(255,255,255,0.1)';ctx.lineWidth=1;ctx.stroke(bodyPath);
      // Accent elements based on group
      ctx.fillStyle = st.accent;
      switch(st.group){
        case 'classic':
          ctx.fillRect(80,60,140,22);
          ctx.fillRect(70,318,160,12);
          const colPath=new Path2D('M110,45 Q150,65 190,45 Q175,80 150,85 Q125,80 110,45 Z');
          ctx.fill(colPath);
          break;
        case 'sport-side':
        case 'wave':
          // Side panels
          const lSlv=new Path2D('M80,70 L40,110 L70,125 L85,85 Z'); ctx.fill(lSlv);
          const rSlv=new Path2D('M220,70 L260,110 L230,125 L215,85 Z'); ctx.fill(rSlv);
          break;
        case 'raglan':
          const lYoke=new Path2D('M80,70 L110,45 L120,90 L85,110 Z'); ctx.fill(lYoke);
          const rYoke=new Path2D('M220,70 L190,45 L180,90 L215,110 Z'); ctx.fill(rYoke);
          const colR=new Path2D('M110,45 Q150,65 190,45 Q175,80 150,85 Q125,80 110,45 Z'); ctx.fill(colR);
          break;
        case 'pinstripe':
          for(let x=100;x<220;x+=18){ctx.beginPath();ctx.moveTo(x,70);ctx.lineTo(x,330);ctx.lineWidth=2;ctx.strokeStyle=st.accent;ctx.stroke();}
          break;
        case 'hourglass':
          ctx.beginPath();ctx.moveTo(120,70);ctx.lineTo(180,70);ctx.lineTo(160,200);ctx.lineTo(180,330);ctx.lineTo(120,330);ctx.lineTo(140,200);ctx.closePath();ctx.fill();
          break;
      }
      // Watermark
      ctx.font='bold 11px serif';ctx.fillStyle='rgba(201,168,124,0.5)';ctx.textAlign='center';
      ctx.fillText(st.id,150,360);
    });
  });
}

function openCustomizer(productId) {
  const p = PRODUCTS.find(x=>x.id===productId); if(!p) return;
  S.product = p;
  S.productType = p.type;
  S.activeGtStyleId = null;
  S.activePrintId = null;
  S.allOverPrintId = null;
  S.zoneColors = { front:'', back:'', leftSleeve:'', rightSleeve:'', collar:'' };
  S.modelLoaded = false;

  // Set initial canvas background
  let initBg = '#C5D3DE';
  if(p.type==='solid' && p.defaultColor) initBg = p.defaultColor;
  else if(p.type==='pattern' && p.gtStyle) {
    const st = GT_STYLES.find(s=>s.id===p.gtStyle);
    if(st) initBg = st.primary;
  } else if(p.colors&&p.colors[0]) initBg = p.colors[0];
  S.canvasBg = initBg;

  // Show customizer
  document.getElementById('page-products').style.display='none';
  document.getElementById('page-customizer').style.display='flex';

  // Update header
  document.getElementById('c-product-name').textContent='— '+p.name;
  document.getElementById('badge-name').textContent = p.name;
  document.getElementById('badge-price').textContent = p.price;
  document.getElementById('design-name-input').value='';

  // Reset fabric canvas
  if(S.fc){
    S.fc.getObjects().slice().forEach(o=>S.fc.remove(o));
    setFabricBg(initBg);
    S.fc.renderAll();
  }
  const bgInp=document.getElementById('canvas-bg-color');if(bgInp)bgInp.value=initBg;

  // Render UI
  renderTabs(p.type);
  renderZoneList();
  renderSizeGrid();
  renderPrintGrid();
  renderGtGroups();
  renderPalette();
  renderPlacement('txt-placement','txtPlacement');
  renderPlacement('logo-placement','logoPlacement');
  updateSummary();
  updateGtActivePanel();

  // Move fabric canvas element into the preview host
  moveFabricToPreview();

  // Init model-viewer
  const ov = document.getElementById('mv-overlay');
  if(ov){ov.style.display='flex';ov.style.opacity='1';}
  initModelViewer();

  // Apply initial style/print
  setTimeout(() => {
    if(p.type==='pattern' && p.gtStyle) {
      const st = GT_STYLES.find(s=>s.id===p.gtStyle);
      if(st) { S.activeGtStyleId=p.gtStyle; applyGtStyle(p.gtStyle, st.primary, st.accent, st.tertiary||null); }
      renderGtGroups();
    } else if(p.type==='solid') {
      setFabricBg(p.defaultColor||'#C5D3DE');
    }
  }, 100);
}

function moveFabricToPreview() {
  // The fabric canvas lives in #fc-offscreen (always mounted).
  // In the canvas tab we move a CLONE into the preview wrapper.
  // Actually, we show the canvas element via CSS transform in the preview host.
  // The host element is already injected in tab-canvas via the #fc-scale-host div.
  const fcEl = document.getElementById('fabric-canvas');
  const host = document.getElementById('fc-scale-host');
  if(fcEl&&host&&!host.contains(fcEl)){
    host.appendChild(fcEl);
  }
}

function showToast(msg, isError=false) {
  let t = document.getElementById('toast');
  if(!t){ t=document.createElement('div'); t.id='toast'; t.style.cssText='position:fixed;bottom:24px;left:50%;transform:translateX(-50%);padding:9px 18px;border-radius:20px;font-size:12px;font-weight:600;font-family:"DM Sans",sans-serif;letter-spacing:0.04em;pointer-events:none;transition:opacity .3s;z-index:9999'; document.body.appendChild(t); }
  t.style.background = isError ? '#c45c5c' : '#2d5a45';
  t.style.color = '#fff';
  t.textContent = msg;
  t.style.opacity='1';
  clearTimeout(t._timer);
  t._timer = setTimeout(()=>{t.style.opacity='0';},2400);
}

// ═══════════════════════════════════════════════════════════════
//  model-viewer SCRIPT LOADER
// ═══════════════════════════════════════════════════════════════

function loadModelViewerScript() {
  return new Promise(res => {
    if(document.querySelector('[data-mv]')){res();return;}
    const s = document.createElement('script');
    s.type='module'; s.setAttribute('data-mv','1');
    s.src='https://ajax.googleapis.com/ajax/libs/model-viewer/3.4.0/model-viewer.min.js';
    s.onload=res; s.onerror=()=>{console.warn('[KaSha] model-viewer failed to load');res();};
    document.head.appendChild(s);
  });
}
