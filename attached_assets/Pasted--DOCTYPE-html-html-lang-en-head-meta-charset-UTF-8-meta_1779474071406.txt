  state.currentViewIdx = 0;

  buildRightPanel();

  setTimeout(() => { if (state.activeTool === 'style') switchTool('color'); }, 600);
}

  img.src = src;
  state.currentViewIdx = viewIdx;
  
  // Update right panel active state
  document.querySelectorAll('.view-btn').forEach((btn, i) => {
    btn.classList.toggle('active', i === viewIdx);
  });
}

function applyColorOverlay() {
  const img = document.getElementById('garmentMainImg');
  const canvas = document.getElementById('tintCanvas');
  if (!img.naturalWidth) return;

  canvas.width = img.offsetWidth;
  canvas.height = img.offsetHeight;
  const ctx = canvas.getContext('2d');
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // Apply the dominant color (full garment if set)
  const color = state.zoneColors.all || state.zoneColors.front;

    ctx.globalAlpha = 0.45;
    ctx.fillRect(0, 0, canvas.width, canvas.height);

  }
}

let layerIdCounter = 0;

function addDesignLayer(config) {
  const id = 'layer_' + (++layerIdCounter);
  const container = document.getElementById('layerContainer');
  const wrap = document.getElementById('garmentImgWrap');
  const wrapRect = wrap.getBoundingClientRect();

  const layer = document.createElement('div');
  layer.className = 'design-layer';
  layer.id = id;
  layer.dataset.layerId = id;

  // Default position based on zone
  const zonePos = getZoneDefaultPos(config.zone || state.activeZone);
  layer.style.cssText = `
    position: absolute;
    left: ${zonePos.x}%;
    top: ${zonePos.y}%;
    transform: translate(-50%, -50%);
    cursor: move;
    user-select: none;
    z-index: 10;
  `;

  if (config.type === 'text') {
    const textEl = document.createElement('div');
    textEl.style.cssText = `
      font-family: ${config.font};
      font-size: ${config.size}px;
      color: ${config.color};
      letter-spacing: ${config.letterSpacing};
      white-space: nowrap;
      text-shadow: 1px 1px 3px rgba(0,0,0,0.3);
      pointer-events: none;
    `;
    textEl.textContent = config.content;
    
    const controls = createLayerControls(id);
    layer.appendChild(controls);
    layer.appendChild(textEl);
  } else if (config.type === 'image') {
    const img = document.createElement('img');
    img.src = config.src;
    img.style.cssText = `width: 120px; height: auto; display: block; pointer-events: none;`;
    
    const controls = createLayerControls(id);
    layer.appendChild(controls);
    layer.appendChild(img);
  }

  // Make draggable
  makeDraggable(layer, container);
  
  container.appendChild(layer);
  state.layers.push({ id, config });
  state.activeLayerId = id;
}

function createLayerControls(id) {
  const ctrl = document.createElement('div');
  ctrl.className = 'layer-controls';
  ctrl.innerHTML = `
    <button class="layer-ctrl-btn" onclick="adjustLayerScale('${id}',1.2)">+</button>
    <button class="layer-ctrl-btn" onclick="adjustLayerScale('${id}',0.8)">−</button>
    <button class="layer-ctrl-btn" onclick="rotateLayer('${id}')">↻</button>
    <button class="layer-ctrl-btn" onclick="removeLayer('${id}')">✕</button>
  `;
  return ctrl;
}

function makeDraggable(el, container) {
  let startX, startY, startLeft, startTop;
  
  el.addEventListener('mousedown', (e) => {
    if (e.target.tagName === 'BUTTON') return;
    e.preventDefault();
    state.activeLayerId = el.id;
    
    const containerRect = container.getBoundingClientRect();
    startX = e.clientX;
    startY = e.clientY;
    
    const computed = getComputedStyle(el);
    startLeft = parseFloat(el.style.left) || 50;
    startTop = parseFloat(el.style.top) || 50;

    function onMove(e) {
      const dx = e.clientX - startX;
      const dy = e.clientY - startY;
      const parentW = container.offsetWidth;
      const parentH = container.offsetHeight;
      el.style.left = (startLeft + (dx / parentW * 100)) + '%';
      el.style.top = (startTop + (dy / parentH * 100)) + '%';
    }
    function onUp() {
      document.removeEventListener('mousemove', onMove);
      document.removeEventListener('mouseup', onUp);
    }
    document.addEventListener('mousemove', onMove);
    document.addEventListener('mouseup', onUp);
  });

  // Touch support
  el.addEventListener('touchstart', (e) => {
    if (e.target.tagName === 'BUTTON') return;
    const touch = e.touches[0];
    startX = touch.clientX; startY = touch.clientY;
    startLeft = parseFloat(el.style.left) || 50;
    startTop = parseFloat(el.style.top) || 50;
    
    function onTouchMove(e) {
      const t = e.touches[0];
      const dx = t.clientX - startX, dy = t.clientY - startY;
      const parentW = container.offsetWidth, parentH = container.offsetHeight;
      el.style.left = (startLeft + (dx / parentW * 100)) + '%';
      el.style.top = (startTop + (dy / parentH * 100)) + '%';
    }
    function onTouchEnd() {
      el.removeEventListener('touchmove', onTouchMove);
      el.removeEventListener('touchend', onTouchEnd);
    }
    el.addEventListener('touchmove', onTouchMove);
    el.addEventListener('touchend', onTouchEnd);
  });
}

function getZoneDefaultPos(zone) {
  const positions = {
    'front-chest': { x: 50, y: 28 },
    'front-center': { x: 50, y: 55 },
    'back-top': { x: 50, y: 28 },
    'back-center': { x: 50, y: 55 },
    'sleeve-left': { x: 12, y: 38 },
    'sleeve-right': { x: 88, y: 38 },
    'collar': { x: 50, y: 12 },
  };
  return positions[zone] || { x: 50, y: 50 };
}

function adjustActiveLayer(prop, val) {
  if (!state.activeLayerId) return;
  const layer = document.getElementById(state.activeLayerId);
  if (!layer) return;
  
  const img = layer.querySelector('img');
  const text = layer.querySelector('div:not(.layer-controls)');
  
  if (prop === 'scale') {
    const scale = val / 100;
    if (img) img.style.width = (120 * scale) + 'px';
    if (text) text.style.fontSize = (parseInt(text.style.fontSize || 42) * scale / (state._lastScale || 1)) + 'px';
    state._lastScale = scale;
  } else if (prop === 'rotate') {
    layer.style.transform = `translate(-50%, -50%) rotate(${val}deg)`;
  } else if (prop === 'opacity') {
    layer.style.opacity = val / 100;
  }
}

function adjustLayerScale(id, factor) {
  const layer = document.getElementById(id);
  if (!layer) return;
  const img = layer.querySelector('img');
  if (img) img.style.width = (parseInt(img.style.width || 120) * factor) + 'px';
  const text = layer.querySelector('div:not(.layer-controls)');
  if (text) text.style.fontSize = (parseInt(text.style.fontSize || 42) * factor) + 'px';
}

function rotateLayer(id) {
  const layer = document.getElementById(id);
  if (!layer) return;
  const current = layer.dataset.rotation ? parseInt(layer.dataset.rotation) : 0;
  const next = current + 15;
  layer.dataset.rotation = next;
  layer.style.transform = `translate(-50%, -50%) rotate(${next}deg)`;
}

function removeLayer(id) {
  const layer = document.getElementById(id);
  if (layer) layer.remove();
  state.layers = state.layers.filter(l => l.id !== id);
  if (state.activeLayerId === id) state.activeLayerId = null;
  pushHistory();
}

// ─── TOOL SWITCHING ───────────────────────────────
function switchTool(tool) {
  state.activeTool = tool;
  
  // Update left nav
  document.querySelectorAll('.tool-btn').forEach(b => b.classList.remove('active'));
  const btn = document.getElementById('tool' + tool.charAt(0).toUpperCase() + tool.slice(1));
  if (btn) btn.classList.add('active');

  // Show/hide panels
  const panels = ['Style','Color','Pattern','Logo','Text','Export'];
  panels.forEach(p => {
    const el = document.getElementById('panel' + p);
    if (el) el.style.display = 'none';
  });
  const activePanel = document.getElementById('panel' + tool.charAt(0).toUpperCase() + tool.slice(1));
  if (activePanel) activePanel.style.display = 'flex';
  activePanel && (activePanel.style.flexDirection = 'column');
  activePanel && (activePanel.style.height = '100%');

  // Update step progress
  const stepMap = { style: 0, color: 1, pattern: 2, logo: 3, text: 4, export: 5 };
  updateStepProgress(stepMap[tool] || 0);
}

function setStep(idx) {
  const tools = ['style','color','pattern','logo','text','export'];
  if (tools[idx]) switchTool(tools[idx]);
}

function updateStepProgress(activeIdx) {
  document.querySelectorAll('.step-item').forEach((item, i) => {
    item.classList.remove('active', 'done');
    if (i < activeIdx) item.classList.add('done');
    else if (i === activeIdx) item.classList.add('active');
  });
}

function pushHistory() {

    productId: state.activeProduct?.id,
    viewIdx: state.currentViewIdx,
    zoneColors: { ...state.zoneColors },
    layers: state.layers.map(l => ({ ...l })),
  };

function undoAction() {
  if (state.history.length <= 1) { showToast('Nothing to undo'); return; }
  state.history.pop();
  const prev = state.history[state.history.length - 1];
  // Restore colors
  state.zoneColors = { ...prev.zoneColors };
  applyColorOverlay();
  showToast('Undone');
}

function exportPNG() {
  showToast('Generating PNG...');
  const img = document.getElementById('garmentMainImg');
  const container = document.getElementById('garmentImgWrap');
  
  // Create a canvas to composite everything
  const canvas = document.createElement('canvas');
  canvas.width = img.naturalWidth || 1200;
  canvas.height = img.naturalHeight || 1500;
  const ctx = canvas.getContext('2d');
  
  // Draw base garment
  const baseImg = new Image();
  baseImg.crossOrigin = 'anonymous';
  baseImg.onload = () => {
    ctx.drawImage(baseImg, 0, 0, canvas.width, canvas.height);
    
    // Apply color overlay
    const color = state.zoneColors.all || state.zoneColors.front;
    if (color) {
      ctx.globalCompositeOperation = 'multiply';
      ctx.fillStyle = color;
      ctx.globalAlpha = 0.45;
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.globalAlpha = 1;
      ctx.globalCompositeOperation = 'source-over';
    }
    
    // Draw text layers
    const textLayers = container.querySelectorAll('.design-layer');
    textLayers.forEach(layer => {
      const textEl = layer.querySelector('div:not(.layer-controls)');
      if (textEl) {
        const containerW = container.offsetWidth;
        const containerH = container.offsetHeight;
        const leftPct = parseFloat(layer.style.left) / 100;
        const topPct = parseFloat(layer.style.top) / 100;
        const x = leftPct * canvas.width;
        const y = topPct * canvas.height;
        ctx.font = textEl.style.fontSize + ' ' + textEl.style.fontFamily;
        ctx.fillStyle = textEl.style.color;
        ctx.textAlign = 'center';
        ctx.fillText(textEl.textContent, x, y);
      }
    });
    
    // Trigger download
    const link = document.createElement('a');
    link.href = canvas.toDataURL('image/png', 0.95);
    link.download = `kasha-design-${state.activeProduct?.id || 'custom'}-${Date.now()}.png`;
    link.click();
    showToast('PNG exported successfully!');
  };
  baseImg.onerror = () => {
    // Fallback: just save the visible img
    const link = document.createElement('a');
    link.href = img.src;
    link.download = `kasha-design-${Date.now()}.png`;
    link.click();
    showToast('Design saved!');
  };
  baseImg.src = img.src;
}
