// ── Color recoloring ─────────────────────────────────────────────────────────
function hexToRgb(hex){
  const r = parseInt(hex.slice(1,3),16);
  const g = parseInt(hex.slice(3,5),16);
  const b = parseInt(hex.slice(5,7),16);
  return {r,g,b};
}
function colorDist(r1,g1,b1,r2,g2,b2){
  return Math.sqrt((r1-r2)**2+(g1-g2)**2+(b1-b2)**2);
}

// Load an image and return a canvas with the image drawn on it
function loadImageData(url){
  return new Promise((res,rej)=>{
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = ()=>{
      const w = img.naturalWidth  || img.width;
      const h = img.naturalHeight || img.height;
      if(!w || !h){ rej(new Error('Image has zero dimensions: '+url)); return; }
      const c = document.createElement('canvas');
      c.width = w; c.height = h;
      c.getContext('2d').drawImage(img, 0, 0);
      res(c);
    };
    img.onerror = ()=>rej(new Error('Failed to load image: '+url));
    img.src = url;
  });
}

// Recolor image: replace SRC_A → cA_hex, SRC_B → cB_hex
// srcCanvas must be an HTMLCanvasElement with non-zero width/height
function recolorImage(srcCanvas, cA_hex, cB_hex){
  const w = srcCanvas.width  | 0;   // bitwise-or 0 converts to integer (guards NaN/undefined)
  const h = srcCanvas.height | 0;
  if(!w || !h) throw new Error('recolorImage: source canvas has zero dimensions');

  const out = document.createElement('canvas');
  out.width = w; out.height = h;
  const ctx = out.getContext('2d');
  ctx.drawImage(srcCanvas, 0, 0);

  const imgData = ctx.getImageData(0, 0, w, h);
  const pix = imgData.data;
  const nA = hexToRgb(cA_hex);
  const nB = hexToRgb(cB_hex);

  for(let i = 0; i < pix.length; i += 4){
    const r = pix[i], g = pix[i+1], b = pix[i+2];
    const dA = colorDist(r, g, b, SRC_A.r, SRC_A.g, SRC_A.b);
    const dB = colorDist(r, g, b, SRC_B.r, SRC_B.g, SRC_B.b);
    if(dA <= dB && dA < TOLERANCE){
      const t = Math.max(0, 1 - dA / TOLERANCE);
      pix[i]   = Math.round(r*(1-t) + nA.r*t) | 0;
      pix[i+1] = Math.round(g*(1-t) + nA.g*t) | 0;
      pix[i+2] = Math.round(b*(1-t) + nA.b*t) | 0;
    } else if(dB < TOLERANCE){
      const t = Math.max(0, 1 - dB / TOLERANCE);
      pix[i]   = Math.round(r*(1-t) + nB.r*t) | 0;
      pix[i+1] = Math.round(g*(1-t) + nB.g*t) | 0;
      pix[i+2] = Math.round(b*(1-t) + nB.b*t) | 0;
    }
  }
  ctx.putImageData(imgData, 0, 0);
  return out;
}

// Cache of loaded source canvases by URL
const srcCache = {};

async function getSrcCanvas(url){
  if(srcCache[url]) return srcCache[url];
  const canvas = await loadImageData(url);   // now returns HTMLCanvasElement directly
  srcCache[url] = canvas;
  return canvas;
}

// ── Baking & applying to 3D model ────────────────────────────────────────────
async function bakeAtlas(){
  // Create the 1024×1024 atlas canvas
  const atlas = document.createElement('canvas');
  atlas.width = ATLAS_W; atlas.height = ATLAS_H;
  const ctx = atlas.getContext('2d');

  // Default background (gray fabric)
  ctx.fillStyle = '#cccccc';
  ctx.fillRect(0,0,ATLAS_W,ATLAS_H);

  // Draw each zone that has been assigned
  for(const zk of ZONE_KEYS){
    const zs  = zoneState[zk];
    if(!zs.patternId) continue;
    const pat = patterns.find(p=>p.id===zs.patternId);
    if(!pat) continue;

    const z = RAW_ZONES[zk];
    let src;
    try{ src = await getSrcCanvas(pat.srcUrl); } catch(e){ console.warn(zk,'src load fail',e); continue; }
    if(!src || !src.width || !src.height){ console.warn(zk,'src canvas zero dims'); continue; }

    // Recolor with the zone's chosen colors
    const recolored = recolorImage(src, zs.colorA, zs.colorB);

    ctx.save();
    ctx.beginPath();
    ctx.rect(z.left, z.top, z.w, z.h);
    ctx.clip();
    ctx.drawImage(recolored, z.left, z.top, z.w, z.h);
    ctx.restore();
  }

  // Show UV preview
  const pvC = document.getElementById('uvCanvas');
  pvC.width = 140; pvC.height = 140;
  pvC.getContext('2d').drawImage(atlas,0,0,140,140);
  document.getElementById('uv-prev').style.display = 'block';

  return atlas;
}

async function applyAtlasToModel(atlas){
  if(!model){ setStatus('⚠ Model not loaded yet'); return; }
  if(compositeTexture) compositeTexture.dispose();
  compositeTexture = new THREE.CanvasTexture(atlas);
  compositeTexture.encoding = THREE.sRGBEncoding;
  compositeTexture.flipY = false;
  compositeTexture.needsUpdate = true;

  model.traverse(node=>{
    if(!node.isMesh) return;
    const mats = Array.isArray(node.material) ? node.material : [node.material];
    mats.forEach(m=>{
      m.map = compositeTexture;
      m.roughness = 0.85;
      m.metalness = 0;
      m.color.set(0xffffff);
      m.needsUpdate = true;
    });
  });
}

  const btn = document.getElementById('btnFull');

}

// Render a recolored thumbnail into an <img> element via data URL
async function renderPreview(p, imgEl){
  try{
    const src = await getSrcCanvas(p.srcUrl);
    if(!src || !src.width || !src.height) return;
    const recolored = recolorImage(src, colorA, colorB);
    imgEl.src = recolored.toDataURL('image/jpeg', 0.85);
  } catch(e){ console.warn('preview fail:', e.message); }
}
