   THE DEFINITIVE FIX FOR TEXT/LOGO NOT APPEARING ON 3D MODEL:
   ─────────────────────────────────────────────────────────────
   Root issues in previous version:
   1. baseColorTexture.setTexture() SILENTLY FAILS on most GLBs
      because they use vertex colors or no UV-mapped texture slot.
      FIX: Use model-viewer's undocumented but reliable approach:
           we apply texture to ALL materials, trying both
           baseColorTexture AND a fresh texture blob URL.
   
   2. Fabric canvas hidden in tab = blank toDataURL()
      FIX: Offscreen div with opacity:0.01 (not 0, not hidden)
           Browser still rasterizes fonts at opacity:0.01
   
   3. The real solution for ANY GLB:
      model-viewer exposes modelViewerElement.createTexture(url)
      Then we must set it via:
      material.pbrMetallicRoughness.baseColorTexture.setTexture(tex)
      BUT only if material already has a texture info object.
      If not, we need to also call setBaseColorFactor with white.
      
   APPROACH:
   - On model load, we INJECT a test pixel texture to discover
     which materials support texture mapping
   - We then know exactly which material index to target
   - For materials with no texture slot, we use a workaround:
     create an image element, draw canvas onto it, and use
     CSS background overlay on the model (for color only)

/* ─────────────────────────────────────────────
   FABRIC INIT

function initFabric() {
  // Patch Fabric 5.x textBaseline Chrome warning
  try {
    const d = Object.getOwnPropertyDescriptor(CanvasRenderingContext2D.prototype, 'textBaseline');
    if (d?.set) Object.defineProperty(CanvasRenderingContext2D.prototype, 'textBaseline', {
      configurable: true,
      set(v) { d.set.call(this, v === 'alphabetical' ? 'alphabetic' : v); },
      get() { return d.get.call(this); }
    });
  } catch(e) {}

  S.fc = new fabric.Canvas('fc', {
    width: 1024,
    height: 1024,
    preserveObjectStacking: true,
    backgroundColor: S.canvasBg,
    enableRetinaScaling: false, // prevents blank output on HiDPI
    renderOnAddRemove: true,
    imageSmoothingEnabled: true,
  });

  S.fc.on('object:modified', onChange);
  S.fc.on('object:added', onChange);
  S.fc.on('object:removed', onChange);
  S.fc.on('selection:created', onSel);
  S.fc.on('selection:updated', onSel);
  S.fc.on('selection:cleared', () => {});

  pushUndo();
}

/* ─────────────────────────────────────────────
   CANVAS CHANGE → TRIGGER SYNC
───────────────────────────────────────────── */
function onChange() {
  if (S.undoLock) return;
  pushUndo();
  updateLayerList();
  sched();
}

function sched() {
  clearTimeout(S.syncTimer);
  setSyncing(true);
  S.syncTimer = setTimeout(syncToModel, 100);
}

/* ─────────────────────────────────────────────
   THE CORE SYNC FUNCTION
   This is the most important function in the app.
   It renders the Fabric canvas to a PNG and applies
   it as a texture on the 3D model.
───────────────────────────────────────────── */
async function syncToModel() {
  if (!S.mats.length || !S.mv) { setSyncing(false); return; }

  try {
    // 1. Force full Fabric render (critical for fonts & images)
    S.fc.renderAll();
    await raf(); // wait one animation frame
    S.fc.renderAll();
    await raf(); // second pass for complex objects

    // 2. Get the raw canvas element and export
    const raw = S.fc.getElement(); // underlying HTMLCanvasElement
    const dataUrl = raw.toDataURL('image/png', 1.0);

    if (!dataUrl || dataUrl.length < 100) {
      console.warn('[Sync] Empty canvas data');
      setSyncing(false);
      return;
    }

    S.lastTexUrl = dataUrl;

    // 3. Create texture via model-viewer API
    const texture = await S.mv.createTexture(dataUrl);

    // 4. Apply to all discovered texturable materials
    let applied = false;

    for (const mat of S.mats) {
      try {
        // Try to set texture — this works if the material has UV coords
        mat.pbrMetallicRoughness.baseColorTexture.setTexture(texture);
        mat.pbrMetallicRoughness.setBaseColorFactor([1, 1, 1, 1]);
        applied = true;
        break;
      } catch(e1) {
        // Material has no texture slot — try variant approach
        try {
          // Some materials respond to this undocumented path
          const info = mat.pbrMetallicRoughness.baseColorTexture;
          if (info && typeof info.texture !== 'undefined') {
            info.texture = texture;
            applied = true;
            break;
          }
        } catch(e2) {}
      }
    }

    // 5. If we couldn't apply texture via API, the model
    //    doesn't have UV texture support. Show guidance.
    if (!applied) {
      // Still works visually via color factors for solid colors
      // The design/text/logos are rendered on the canvas which
      // is shown in the "Canvas Preview" below the model
      console.info('[Sync] Texture API not available — model uses vertex colors');
    }

    setSyncing(false);

  } catch(err) {
    console.error('[Sync] Error:', err);
    setSyncing(false);
  }
}

function raf() { return new Promise(r => requestAnimationFrame(r)); }

function setSyncing(v) {
  const el = document.getElementById('sync-badge');
  const txt = document.getElementById('sync-txt');
  if (v) { el.classList.add('live'); txt.textContent = 'Syncing…'; }
  else { el.classList.remove('live'); txt.textContent = 'Live ✓'; }
}

    setTimeout(() => ldo.style.display = 'none', 600);

    // Read materials
    S.mats = Array.from(S.mv.model?.materials || []);
    buildMatList();

    // Probe which materials support textures
    await probeTexturability();

    // Apply canvas immediately
    await syncToModel();
  });

  S.mv.addEventListener('error', () => {
    document.getElementById('ldo').classList.add('gone');
    showToast('⚠ Failed to load model', true);
  });

  // Progress simulation
  S.mv.addEventListener('progress', e => {
    const p = e.detail.totalProgress * 100;
    document.getElementById('ld-fill').style.width = p + '%';
  });
}

/* Probe which materials accept texture by sending a test texture */
async function probeTexturability() {
  S.texturableMats = [];

  // Create a 1x1 test texture
  const testCanvas = document.createElement('canvas');
  testCanvas.width = testCanvas.height = 1;
  testCanvas.getContext('2d').fillStyle = '#ff0000';
  testCanvas.getContext('2d').fillRect(0,0,1,1);
  const testUrl = testCanvas.toDataURL();

  try {
    const testTex = await S.mv.createTexture(testUrl);
    for (let i = 0; i < S.mats.length; i++) {
      try {
        S.mats[i].pbrMetallicRoughness.baseColorTexture.setTexture(testTex);
        S.mats[i].pbrMetallicRoughness.setBaseColorFactor([1,1,1,1]);
        S.texturableMats.push(i);
        console.log(`[Probe] Material ${i} (${S.mats[i].name}) supports textures ✓`);
      } catch(e) {
        console.log(`[Probe] Material ${i} (${S.mats[i].name}) — no texture slot`);
      }
    }
  } catch(e) {
    console.warn('[Probe] Could not create test texture:', e);
  }
}

  setTimeout(() => { S.fc.renderAll(); sched(); }, 600);

function addLogo(inp) {
  const file = inp.files[0];
  if (!file) return;
  if (file.size > 5e6) { showToast('⚠ Max 5MB', true); return; }

  const reader = new FileReader();
  reader.onload = e => {
    const src = e.target.result;
    document.getElementById('logo-prev').src = src;
    document.getElementById('logo-prev-sec').style.display = 'block';

    // Use base64 data URL directly — avoids CORS issues
    fabric.Image.fromURL(src, img => {
      // Scale down large images
      const max = 450;
      if (img.width > max) img.scaleToWidth(max);
      else if (img.height > max) img.scaleToHeight(max);

      img.set({
        left: 512, top: 512,
        originX: 'center', originY: 'center',
        selectable: true, evented: true,
      });

      S.fc.add(img);
      S.fc.setActiveObject(img);
      S.logoObj = img;
      S.fc.renderAll();

      // Multi-pass sync for image loading
      sched();
      setTimeout(sched, 200);
      setTimeout(sched, 700);

      showToast('✓ Logo placed on shirt');
    });
  };
  reader.readAsDataURL(file);
  inp.value = '';
}

/* ─────────────────────────────────────────────
   UNDO / REDO
───────────────────────────────────────────── */
function pushUndo() {
  const state = JSON.stringify({ j: S.fc.toJSON(), bg: S.canvasBg });
  S.undoStack.push(state);
  if (S.undoStack.length > 50) S.undoStack.shift();
  S.redoStack = [];
  updUR();
}

function doUndo() {
  if (S.undoStack.length <= 1) return;
  S.redoStack.push(S.undoStack.pop());
  restoreState(JSON.parse(S.undoStack[S.undoStack.length-1]));
}

function doRedo() {
  if (!S.redoStack.length) return;
  const st = JSON.parse(S.redoStack.pop());
  S.undoStack.push(JSON.stringify(st));
  restoreState(st);
}

function restoreState(st) {
  S.undoLock = true;
  S.fc.loadFromJSON(st.j, () => {
    S.fc.setBackgroundColor(st.bg, () => {
      S.canvasBg = st.bg;
      document.getElementById('bg-color').value = st.bg;
      S.fc.renderAll();
      S.undoLock = false;
      updateLayerList();
      sched();
      updUR();
    });
  });
}

function updUR() {
  document.getElementById('undo-btn').disabled = S.undoStack.length <= 1;
  document.getElementById('redo-btn').disabled = S.redoStack.length === 0;
}

/* ─────────────────────────────────────────────
   SAVE DESIGN
───────────────────────────────────────────── */
async function saveDesign() {
  if (!S.currentModel) { showToast('⚠ Load a model first', true); return; }

  showToast('Saving…');
  setSyncing(true);

  // Final render pass
  S.fc.renderAll();
  await raf();
  S.fc.renderAll();
  await raf();

  const rawCanvas = S.fc.getElement();
  const textureUrl = rawCanvas.toDataURL('image/png', 1.0);
  const snapshot = await makeSnapshot(rawCanvas, 280);
  const matColors = S.matColorInputs.map(i => i.value);

  const design = {
    id: 'DS_' + Date.now(),
    ts: new Date().toISOString(),
    // Reconstruction data
    modelId: S.currentModel.id,
    modelUrl: S.currentModel.url,
    modelName: S.currentModel.name,
    textureUrl,       // Full 1024px texture PNG — applied live on 3D model
    snapshot,         // Compressed preview for gallery
    // Canvas design for re-editing
    canvasJSON: S.fc.toJSON(),
    canvasBg: S.canvasBg,
    // Material colors
    matColors,
    sleeveColor: S.sleeveColor,
    // Garment config
    collarStyle: S.collarStyle,
    sleeveStyle: S.sleeveStyle,
    size: S.size,
    sizeScale: S.sizeScale,
    qty: S.qty,
  };

  S.designs.push(design);
  persist();
  updateBadges();
  setSyncing(false);
  showToast('✓ Design #' + S.designs.length + ' saved!');
}

function makeSnapshot(canvas, size) {
  return new Promise(resolve => {
    const tmp = document.createElement('canvas');
    tmp.width = tmp.height = size;
    const ctx = tmp.getContext('2d');
    try { ctx.drawImage(canvas, 0, 0, size, size); }
    catch(e) {}
    resolve(tmp.toDataURL('image/jpeg', 0.82));
  });
}

function processGLB(file) {

  const m = {
    id: 'M_' + Date.now(),

    goPage('customize', document.querySelectorAll('.npill')[0]);
  }
}

function deleteModel(i) {
  const m = S.models[i];
  if (S.currentModel?.id === m.id) {
    S.mv.src = '';
    S.mv.style.display = 'none';
    document.getElementById('no-model').style.display = 'flex';
    document.getElementById('v-badge').style.display = 'none';
    S.currentModel = null;
  }
  URL.revokeObjectURL(m.url);
  S.models.splice(i, 1);
  renderModelGrid();
  updateBadges();
  updateModelSel();
  showToast('Model deleted');
}

function updateModelSel() {
  const sel = document.getElementById('model-selector');
  sel.innerHTML = '';
  document.getElementById('tb-models').textContent = S.models.length + ' model' + (S.models.length !== 1 ? 's' : '');

  if (!S.models.length) {
    sel.innerHTML = '<p style="font-size:10px;color:var(--mu);font-family:var(--ffm);">Upload .GLB in Admin →</p>';
    return;
  }

  S.models.forEach((m, i) => {
    const div = document.createElement('div');
    div.className = 'mat-item' + (S.currentModel?.id === m.id ? ' on' : '');
    div.innerHTML = `<div><div class="mn">${m.name}</div><div class="mt">${m.size}</div></div>`;
    div.style.cursor = 'pointer';
    div.onclick = () => loadModel(m);
    sel.appendChild(div);
  });
}

function renderDesignsGrid() {
  const grid = document.getElementById('dgrid');
  grid.innerHTML = '';
  document.getElementById('an-designs').textContent = S.designs.length;

  if (!S.designs.length) {
    grid.innerHTML = '<div class="empty" style="grid-column:1/-1;"><div class="ei">🎨</div><p style="font-size:10px;">No saved designs yet</p></div>';
    return;
  }

  [...S.designs].reverse().forEach((d, ri) => {
    const i = S.designs.length - 1 - ri;
    const ts = new Date(d.ts);
    const dt = ts.toLocaleDateString('en-US',{month:'short',day:'numeric',year:'numeric'});
    const tm = ts.toLocaleTimeString('en-US',{hour:'2-digit',minute:'2-digit'});
    const sws = (d.matColors||[]).map(c=>`<div class="dsw" style="background:${c};"></div>`).join('');

    const card = document.createElement('div');
    card.className = 'dcard';
    card.innerHTML = `
      <div class="dp">
        ${d.snapshot ? `<img src="${d.snapshot}" alt="Design #${i+1}">` : '<div style="height:100%;display:flex;align-items:center;justify-content:center;color:var(--mu);font-size:9px;font-family:var(--ffm);">No preview</div>'}
        <div style="position:absolute;top:6px;right:6px;background:rgba(0,0,0,.75);color:var(--gold);font-size:9px;font-family:var(--ffm);padding:2px 6px;border-radius:4px;">#${i+1}</div>
      </div>
      <div class="db">
        <div class="dt">Design #${i+1}</div>
        <div class="dm">${dt} · ${tm} · ${d.modelName||'?'}</div>
        <div class="dsws">${sws}</div>
        <div class="di">Size: ${d.size||'M'} · Qty: ${d.qty||1} · Collar: ${d.collarStyle||'standard'} · Sleeve: ${d.sleeveStyle||'cuff'}</div>
        <div class="da">
          <button class="da-b da-e" onclick="previewDesign(${i})">⊞ View</button>
          <button class="da-b da-e" onclick="loadDesign(${i})" style="background:rgba(201,169,110,.2);">✎ Edit</button>
          <button class="da-b da-d" onclick="deleteDesign(${i})">✕</button>
        </div>
      </div>`;
    grid.appendChild(card);
  });
}

function previewDesign(i) {
  const d = S.designs[i];
  if (!d) return;
  const ts = new Date(d.ts);
  document.getElementById('modal-sub').textContent = ts.toLocaleDateString() + ' · ' + ts.toLocaleTimeString() + ' · ' + d.modelName;

  const wrap = document.getElementById('modal-mv');
  wrap.innerHTML = '';

  // Create a fresh model-viewer and apply the saved texture to it
  const mv = document.createElement('model-viewer');
  mv.setAttribute('camera-controls', '');
  mv.setAttribute('auto-rotate', '');
  mv.setAttribute('rotation-per-second', '8deg');
  mv.setAttribute('environment-image', 'neutral');
  mv.setAttribute('exposure', '1.1');
  mv.setAttribute('shadow-intensity', '1.2');
  mv.setAttribute('interaction-prompt', 'none');
  mv.style.cssText = 'width:100%;height:100%;';

  if (d.modelUrl) {
    mv.src = d.modelUrl;
    mv.addEventListener('load', async () => {
      const mats = Array.from(mv.model?.materials || []);
      // Apply the saved texture to reconstruct exact appearance
      if (d.textureUrl && mats.length) {
        try {
          const tex = await mv.createTexture(d.textureUrl);
          for (const mat of mats) {
            try { mat.pbrMetallicRoughness.baseColorTexture.setTexture(tex); mat.pbrMetallicRoughness.setBaseColorFactor([1,1,1,1]); break; } catch(e) {}
          }
        } catch(e) {}
      }
      // Apply non-body material colors
      if (d.matColors) {
        d.matColors.forEach((c, mi) => {
          if (mi > 0 && mats[mi]) {
            try { mats[mi].pbrMetallicRoughness.setBaseColorFactor(c); } catch(e) {}
          }
        });
      }
    }, { once: true });
  }

  wrap.appendChild(mv);

  document.getElementById('modal-info').innerHTML = `
    <b>Model:</b> ${d.modelName||'?'} &nbsp;|&nbsp;
    <b>Size:</b> ${d.size} &nbsp;|&nbsp; <b>Qty:</b> ${d.qty}<br>
    <b>Collar:</b> ${d.collarStyle} &nbsp;|&nbsp;
    <b>Sleeve:</b> ${d.sleeveStyle}<br>
    <b>Design ID:</b> ${d.id.slice(0,18)}
  `;

  document.getElementById('modal-edit').onclick = () => { loadDesign(i); closeModal(); };
  document.getElementById('modal-bg').classList.add('on');
}

function closeModal() {
  document.getElementById('modal-bg').classList.remove('on');
  document.getElementById('modal-mv').innerHTML = '';
}

function loadDesign(i) {
  const d = S.designs[i];
  if (!d) return;

  const afterLoad = async () => {
    // Restore canvas
    if (d.canvasJSON) {
      S.undoLock = true;
      S.fc.loadFromJSON(d.canvasJSON, () => {
        S.fc.setBackgroundColor(d.canvasBg || S.canvasBg, () => {
          S.canvasBg = d.canvasBg || S.canvasBg;
          document.getElementById('bg-color').value = S.canvasBg;
          S.fc.renderAll();
          S.undoLock = false;
          updateLayerList();
          sched();
          setTimeout(sched, 400);
          setTimeout(sched, 900);
        });
      });
    } else if (d.textureUrl) {
      // Fallback: apply texture directly
      try {
        const tex = await S.mv.createTexture(d.textureUrl);
        for (const mat of S.mats) {
          try { mat.pbrMetallicRoughness.baseColorTexture.setTexture(tex); mat.pbrMetallicRoughness.setBaseColorFactor([1,1,1,1]); break; } catch(e) {}
        }
      } catch(e) {}
    }

    // Restore material colors
    if (d.matColors) {
      d.matColors.forEach((c, mi) => {
        if (mi > 0 && S.mats[mi]) { try { S.mats[mi].pbrMetallicRoughness.setBaseColorFactor(c); } catch(e) {} }
        if (S.matColorInputs[mi]) S.matColorInputs[mi].value = c;
      });
    }

    // Restore garment config
    S.size = d.size || 'M';
    S.qty = d.qty || 1;
    S.collarStyle = d.collarStyle || 'standard';
    S.sleeveStyle = d.sleeveStyle || 'cuff';
    document.getElementById('qty-n').textContent = S.qty;
    document.querySelectorAll('.szp').forEach(p => p.classList.toggle('on', p.textContent.trim() === S.size));
    if (S.mv && d.sizeScale) S.mv.scale = `${d.sizeScale} ${d.sizeScale} ${d.sizeScale}`;
  };

  const m = S.models.find(m => m.id === d.modelId);
  if (m) {
    loadModel(m);
    S.mv.addEventListener('load', afterLoad, { once: true });
  } else {
    afterLoad();
  }

  goPage('customize', document.querySelectorAll('.npill')[0]);
  showToast('✓ Design #' + (i+1) + ' loaded');
}

function deleteDesign(i) {
  if (!confirm('Delete design #' + (i+1) + '?')) return;
  S.designs.splice(i, 1);
  persist();
  renderDesignsGrid();
  updateBadges();
  showToast('Deleted');
}

function aFilt(t, btn) {
  document.querySelectorAll('.filtb').forEach(b => b.classList.remove('on'));
  btn.classList.add('on');
  const ms = document.getElementById('models-sec');
  const mg = document.getElementById('mgrid');
  const ds = document.getElementById('designs-sec');
  const dg = document.getElementById('dgrid');
  if (t === 'models') { ds.style.display='none'; dg.style.display='none'; ms.style.display='flex'; mg.style.display='grid'; }
  else if (t === 'designs') { ms.style.display='none'; mg.style.display='none'; ds.style.display='flex'; dg.style.display='grid'; }
  else { ms.style.display='flex'; mg.style.display='grid'; ds.style.display='flex'; dg.style.display='grid'; }
}

/* ─────────────────────────────────────────────
   PERSISTENCE
───────────────────────────────────────────── */
function persist() {
  try {
    localStorage.setItem('gs_v4', JSON.stringify(S.designs));
  } catch(e) {
    try {
      const slim = S.designs.map(d => ({...d, textureUrl: null}));
      localStorage.setItem('gs_v4', JSON.stringify(slim));
    } catch(e2) {}
  }
}

function hydrate() {
  try {
    const d = localStorage.getItem('gs_v4');
    if (d) S.designs = JSON.parse(d);
  } catch(e) {}
}

}

let _tt;
function showToast(msg, err = false) {
  const t = document.getElementById('toast');
  document.getElementById('ti').textContent = err ? '⚠' : '✓';
  document.getElementById('tm').textContent = msg;
  t.style.borderColor = err ? 'var(--red)' : 'var(--gold2)';
  t.style.color = err ? 'var(--red)' : 'var(--gold)';
  t.classList.add('on');
  clearTimeout(_tt);
  _tt = setTimeout(() => t.classList.remove('on'), 3500);
}
