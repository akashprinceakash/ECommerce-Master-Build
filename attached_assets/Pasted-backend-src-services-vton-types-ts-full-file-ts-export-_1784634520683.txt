export type GarmentRole = "top" | "bottom" | "dress";

export const ROLE_TO_VTON_CATEGORY: Record<GarmentRole, "upper_body" | "lower_body" | "dresses"> = {
  top: "upper_body",
  bottom: "lower_body",
  dress: "dresses",
};

/**
 * Per-category IDM-VTON tuning. Left at low step counts with crop off, the
 * model defaults to "safe" full-length garments (e.g. rendering shorts as
 * full trousers), since it was trained mostly on full-length lower-body
 * references. Higher steps + crop=true for lower_body meaningfully improves
 * garment-length fidelity (shorts/skorts staying short instead of extending).
 */
export const VTON_CATEGORY_TUNING: Record<"upper_body" | "lower_body" | "dresses", { steps: number; crop: boolean }> = {
  upper_body: { steps: 30, crop: false },
  lower_body: { steps: 35, crop: true },
  dresses: { steps: 30, crop: false },
};

/**
 * Maps raw catalog categories to an explicit length/style hint appended to
 * the garment description sent to IDM-VTON. This is the main lever for
 * getting shorts/skorts to render short instead of as full trousers, and
 * for keeping dresses/skirts distinct from pants. Keys must match the
 * category strings used by classifyProductRole / the frontend's
 * TOP_CATEGORIES / BOTTOM_CATEGORIES / DRESS_CATEGORIES sets.
 */
const CATEGORY_LENGTH_HINTS: Record<string, string> = {
  // Tops
  "t-shirt": "a short-sleeve t-shirt",
  "polo": "a short-sleeve polo shirt with a collar",
  "fabric-tshirt": "a short-sleeve t-shirt",
  "pattern": "a patterned short-sleeve shirt",
  "shirts": "a shirt",
  // Bottoms
  "pants": "full-length trousers reaching the ankle",
  "trousers": "full-length tailored trousers reaching the ankle",
  "shorts": "short shorts ending above the knee — NOT full-length trousers",
  "skort": "a short skort (skirt-shorts hybrid) ending above the knee — NOT full-length trousers",
  "skorts": "a short skort (skirt-shorts hybrid) ending above the knee — NOT full-length trousers",
  "skirts": "a skirt — NOT trousers or pants",
  // Dresses
  "dress": "a full dress",
  "dresses": "a full dress",
  "golf dress": "a golf dress",
  "golf dresses": "a golf dress",
};

/**
 * Build the `garment_des` text prompt sent to IDM-VTON. Combines the
 * product's display name with an explicit length/style hint derived from
 * its raw catalog category — IDM-VTON leans on this text to decide garment
 * length/silhouette, and otherwise defaults toward trousers.
 */
export function buildGarmentDescription(name: string, category: string): string {
  const hint = CATEGORY_LENGTH_HINTS[category.trim().toLowerCase()];
  return hint ? `${name}, ${hint}` : name;
}

export interface TryOnGarment {
  productId: number;
  role: GarmentRole;
  name: string;
  imageUrl: string;
  /** Raw catalog category (e.g. "shorts", "skort", "pants") — used to build
   *  a length/style-aware garment_des so IDM-VTON doesn't default to
   *  full-length trousers for short garments. */
  category: string;
}

export interface StartPredictionParams {
  humanImageUrl: string;
  garmentImageUrl: string;
  garmentDescription: string;
  vtonCategory: "upper_body" | "lower_body" | "dresses";
  /** Optional overrides — default to the per-category tuning in types.ts. */
  steps?: number;
  crop?: boolean;
}

export async function startIdmVtonPrediction(
  params: StartPredictionParams,
): Promise<string> {
  const { humanImageUrl, garmentImageUrl, garmentDescription, vtonCategory } = params;
  const tuning = VTON_CATEGORY_TUNING[vtonCategory];
  const steps = params.steps ?? tuning.steps;
  const crop = params.crop ?? tuning.crop;

  const version = await getModelVersionId();
  const prediction = await replicateFetch<ReplicatePrediction>("/predictions", {
    method: "POST",
    body: JSON.stringify({
      version,
      input: {
        human_img: humanImageUrl,
        garm_img: garmentImageUrl,
        garment_des: garmentDescription,
        category: vtonCategory,
        crop,
        force_dc: vtonCategory === "dresses",
        steps,
        seed: Math.floor(Math.random() * 1_000_000),
      },
    }),
  });
  logger.info({ predictionId: prediction.id, vtonCategory, steps, crop }, "vton: prediction started");
  return prediction.id;
}

What this does and doesn't guarantee
Applies to every combination you asked about — t-shirt/polo + pants, + shorts, + skort, or a standalone full dress — since buildGarmentDescription covers all those categories and the tuning map applies per VTON category (upper_body/lower_body/dresses), not per specific product.
Works the same for a client's own uploaded photo or the AI avatar — the fix is in how the garment is described and processed, not in which human image is used, so it applies uniformly to humanImageUrl from /api/lookbook-photo or the stock avatar.
What it fixes: the model previously had almost no signal that "Women - Golf - Skorts" ≠ full trousers, and ran the second-step inference at lower step count without crop. This directly targets the "shorts became long pink pants" failure.
What it doesn't fully eliminate: for two-item outfits (top+bottom), step 2 still runs IDM-VTON against the synthetic output of step 1 rather than the original photo (the currentHumanImage = resultUrl chaining). That remains a real, if reduced, risk of color bleed or misparsing on some photos/poses. If you still see occasional issues after this deploy, the next-level fix is running both garments independently against the original photo and compositing the two results at the waistline — that's a bigger change (needs image compositing with sharp, and reliable waist-position handling across arbitrary user photos) and I'd want to test the above first before investing in it.

Option B — Fragile stopgap using name keyword matching, so the 4 existing polos work today and it degrades gracefully (returns null → "not yet supported" — same as now) if a future q_club pant doesn't match a known keyword, rather than silently misclassifying it:

ts
import type { GarmentRole } from "./types";

const TOP_CATEGORIES = new Set(["t-shirt", "polo", "fabric-tshirt", "pattern", "shirts"]);
const BOTTOM_CATEGORIES = new Set(["pants", "trousers", "shorts", "skort", "skorts", "skirts"]);
const DRESS_CATEGORIES = new Set(["dress", "dresses", "golf dress", "golf dresses"]);

/**
 * "q_club" is a bulk-order/no-customization catalog flag, not a garment
 * type — it can hold t-shirts, polos, or pants, and neither `product_type`
 * nor `sub_type` reliably distinguishes them (as of writing, sub_type is
 * just a print style like "printed"). Until product_type is populated for
 * q_club uploads, we fall back to keyword-matching the product name.
 * This is intentionally conservative: an unmatched name returns null
 * (same "not yet supported" behavior as before) rather than guessing.
 */
const QCLUB_TOP_KEYWORDS = ["polo", "t-shirt", "tshirt", "shirt"];
const QCLUB_BOTTOM_KEYWORDS = ["trouser", "pant", "short", "skort"];

function classifyQClubByName(name: string | null | undefined): GarmentRole | null {
  const n = (name ?? "").toLowerCase();
  if (QCLUB_TOP_KEYWORDS.some(k => n.includes(k))) return "top";
  if (QCLUB_BOTTOM_KEYWORDS.some(k => n.includes(k))) return "bottom";
  return null;
}

/**
 * Classify a product's DB `category` into a try-on garment role.
 * Returns null for categories not yet supported by the try-on pipeline
 * (e.g. outerwear like "jacket", accessories, shoes, bags).
 */
export function classifyProductRole(
  category: string | null | undefined,
  name?: string | null,
): GarmentRole | null {
  const c = (category ?? "").toLowerCase().trim();
  if (TOP_CATEGORIES.has(c)) return "top";
  if (BOTTOM_CATEGORIES.has(c)) return "bottom";
  if (DRESS_CATEGORIES.has(c)) return "dress";
  if (c === "q_club") return classifyQClubByName(name);
  return null;
}
