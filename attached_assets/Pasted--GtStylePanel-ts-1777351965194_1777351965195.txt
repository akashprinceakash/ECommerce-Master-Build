const GT_GROUPS = [
  { id:"classic",    label:"Classic"     },
  { id:"sport-side", label:"Sport Side"  },
  { id:"triple",     label:"Triple Tone" },
  { id:"wave",       label:"Wave Panel"  },
  { id:"hourglass",  label:"Hourglass"   },
  { id:"pinstripe",  label:"Pinstripe"   },
  { id:"raglan",     label:"Raglan"      },
];

const BRAND_PALETTE = [
  "#C5D3DE","#F8F4E9","#ACB1A1","#F0CED2","#E9DAC3",
  "#FFFFFF","#585858","#576043","#DA1F26","#273878",

// ── Canvas swatch thumbnail per style ────────────────────────────────────────
function GtSwatch({ style, active }: { style: GtStyleDef; active: boolean }) {
  const ref = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const el = ref.current; if (!el) return;
    const ctx = el.getContext("2d"); if (!ctx) return;
    const W = 56, H = 56;
    const { primary: P, accent: A, tertiary: T } = style.defaultColors;
    ctx.clearRect(0, 0, W, H);

    // Body fill
    ctx.fillStyle = P;
    ctx.fillRect(0, 0, W, H);

    switch (style.group) {

      case "classic":
        // Collar band top
        ctx.fillStyle = A;
        ctx.fillRect(0, 0, W, 10);
        // Cuff band bottom
        ctx.fillRect(0, H - 9, W, 9);
        // Hem line
        ctx.fillRect(0, H - 14, W, 4);
        break;

      case "sport-side":
      case "wave": {
        // Tapered side panels
        const topW = style.group === "wave" ? W*0.32 : W*0.28;
        const botW = style.group === "wave" ? W*0.22 : W*0.18;
        ctx.fillStyle = A;
        ctx.beginPath();
        ctx.moveTo(0,0); ctx.lineTo(topW,0); ctx.lineTo(botW,H); ctx.lineTo(0,H);
        ctx.closePath(); ctx.fill();
        ctx.beginPath();
        ctx.moveTo(W,0); ctx.lineTo(W-topW,0); ctx.lineTo(W-botW,H); ctx.lineTo(W,H);
        ctx.closePath(); ctx.fill();
        break;
      }

      case "triple": {
        const topW = W*0.28, botW = W*0.18;
        ctx.fillStyle = A;
        ctx.beginPath();
        ctx.moveTo(0,0); ctx.lineTo(topW,0); ctx.lineTo(botW,H); ctx.lineTo(0,H);
        ctx.closePath(); ctx.fill();
        ctx.beginPath();
        ctx.moveTo(W,0); ctx.lineTo(W-topW,0); ctx.lineTo(W-botW,H); ctx.lineTo(W,H);
        ctx.closePath(); ctx.fill();
        if (T) { ctx.fillStyle = T; ctx.fillRect(0,0,W,9); }
        break;
      }

      case "hourglass": {
        const cx = W/2;
        ctx.fillStyle = A;
        ctx.beginPath();
        ctx.moveTo(cx-W*0.46, 0);
        ctx.lineTo(cx+W*0.46, 0);
        ctx.lineTo(cx+W*0.20, H*0.52);
        ctx.lineTo(cx+W*0.46, H);
        ctx.lineTo(cx-W*0.46, H);
        ctx.lineTo(cx-W*0.20, H*0.52);
        ctx.closePath(); ctx.fill();
        break;
      }

      case "pinstripe": {
        ctx.strokeStyle = A; ctx.lineWidth = 2;
        const lx = W*0.26, rx = W*0.74;
        ctx.beginPath(); ctx.moveTo(lx,0); ctx.lineTo(lx,H); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(rx,0); ctx.lineTo(rx,H); ctx.stroke();
        // Oval hints
        ctx.beginPath(); ctx.ellipse(lx, H*0.68, W*0.06, H*0.11, 0, 0, Math.PI*2); ctx.stroke();
        ctx.beginPath(); ctx.ellipse(rx, H*0.68, W*0.06, H*0.11, 0, 0, Math.PI*2); ctx.stroke();
        break;
      }

      case "raglan": {
        ctx.fillStyle = A;
        // Left diagonal yoke
        ctx.beginPath();
        ctx.moveTo(0,0); ctx.lineTo(W*0.42,0);
        ctx.lineTo(W*0.16, H*0.26); ctx.lineTo(0, H*0.15);
        ctx.closePath(); ctx.fill();
        // Right diagonal yoke
        ctx.beginPath();
        ctx.moveTo(W,0); ctx.lineTo(W*0.58,0);
        ctx.lineTo(W*0.84, H*0.26); ctx.lineTo(W, H*0.15);
        ctx.closePath(); ctx.fill();
        // Collar
        ctx.fillRect(0, 0, W, 8);
        // Cuff
        ctx.fillRect(0, H-9, W, 9);
        break;
      }
    }

    // Active border
    if (active) {
      ctx.strokeStyle = "#6ee7b7"; ctx.lineWidth = 3;
      ctx.strokeRect(1.5, 1.5, W-3, H-3);
    }
  }, [style, active]);

  return (
    <canvas ref={ref} width={56} height={56}
      style={{ display:"block", width:"100%", height:"auto" }} />
  );
}
