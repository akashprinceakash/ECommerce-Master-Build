/* ── FABRIC INIT ── */
function initFabric() {
  // Patch Chrome textBaseline warning
  try {
    const d = Object.getOwnPropertyDescriptor(CanvasRenderingContext2D.prototype,'textBaseline');
    if (d && d.set) Object.defineProperty(CanvasRenderingContext2D.prototype,'textBaseline',{configurable:true,set(v){d.set.call(this,v==='alphabetical'?'alphabetic':v);},get(){return d.get.call(this);}});
  } catch(e){}

  fC = new fabric.Canvas('fc', { width:1024, height:1024, preserveObjectStacking:true, backgroundColor:'#FFFFFF' });

  const scale = () => {
    const el = document.getElementById('co');
    if (!el || !el.clientWidth) return;
    const w = el.clientWidth;
    const z = w / 1024;
    fC.setZoom(z);
    fC.setWidth(w);
    fC.setHeight(w);
  };
  window.addEventListener('resize', scale);
  setTimeout(scale, 200);

  fC.on('object:modified', sync);
  fC.on('object:added', sync);
  fC.on('object:removed', sync);
  fC.on('selection:created', onSel);
  fC.on('selection:updated', onSel);
}

function onSel(e) {
  const o = e.selected && e.selected[0];
  if (!o) return;
  document.getElementById('el-s').value = o.scaleX || 1;
  document.getElementById('el-sv').textContent = parseFloat(o.scaleX||1).toFixed(1)+'×';
  document.getElementById('el-x').value = Math.round(o.left||512);
  document.getElementById('el-y').value = Math.round(o.top||512);
}

/* ── MODEL LOAD ── */
mv.addEventListener('load', () => {
  const ov = document.getElementById('ldo');
  ov.style.opacity = '0';
  setTimeout(() => ov.style.display = 'none', 500);
  mats = mv.model?.materials || [];
  buildMatList();
  sync();
});

function buildMatList() {
  const list = document.getElementById('mat-list');
  list.innerHTML = '';
  if (!mats.length) { list.innerHTML = '<p style="font-size:11px;color:var(--mu);">No parts found.</p>'; return; }

  const tagNames = ['Main body','Trim / collar','Sleeve panel','Buttons','Detail'];
  mats.forEach((mat, i) => {
    const div = document.createElement('div');
    div.className = 'mi';
    div.dataset.i = i;

    const info = document.createElement('div');
    const nm = document.createElement('div'); nm.className = 'mn'; nm.textContent = mat.name || `Part ${i+1}`;
    const tg = document.createElement('div'); tg.className = 'mt'; tg.textContent = tagNames[i] || `Part ${i+1}`;
    info.appendChild(nm); info.appendChild(tg);

    const cp = document.createElement('input');
    cp.type = 'color'; cp.value = '#ffffff';
    cp.style.cssText = 'width:30px;height:24px;border:none;cursor:pointer;background:none;border-radius:5px;';
    cp.addEventListener('input', e => {
      e.stopPropagation();
      applyColorToMat(i, e.target.value);
    });

    div.addEventListener('click', () => {
      document.querySelectorAll('.mi').forEach(d => d.classList.remove('on'));
      div.classList.add('on');
      activeMat = mat;
      activeMatIdx = i;
      document.getElementById('cp-custom').value = cp.value;
    });

    div.appendChild(info);
    div.appendChild(cp);
    list.appendChild(div);
  });

  // Select first
  list.querySelector('.mi')?.click();
}

function applyColorToMat(idx, hex) {
  const mat = mats[idx];
  if (!mat) return;
  if (idx === 0) {
    S.primaryColor = hex;
    setCanvasBg(hex);
    document.getElementById('cbg').value = hex;
    document.getElementById('cp-custom').value = hex;
  } else {
    mat.pbrMetallicRoughness.setBaseColorFactor(hex);
    if (idx === 1) S.secondaryColor = hex;
  }
}

/* Custom color picker -> active part */
document.getElementById('cp-custom').addEventListener('input', e => {
  const activeDiv = document.querySelector('.mi.on');
  const idx = activeDiv ? parseInt(activeDiv.dataset.i) : 0;
  applyColorToMat(idx, e.target.value);
  const cp = activeDiv?.querySelector('input[type=color]');
  if (cp) cp.value = e.target.value;
});

/* ── PALETTE ── */

  const row = document.getElementById('pal-row');
  PAL.forEach(hex => {
    const s = document.createElement('div');
    s.className = 'sw' + (hex === '#FFFFFF' ? ' wh' : '');
    s.style.background = hex;
    s.title = hex;
    s.onclick = () => {

      applyColorToMat(idx, hex);
      const cp = activeDiv?.querySelector('input[type=color]');

  });
})();

/* ── TEXTURE SYNC ── */
async function sync() {
  if (!mats.length) return;
  try {
    const url = fC.toDataURL({ format:'png', quality:0.92, multiplier:1 });
    const tex = await mv.createTexture(url);
    mats[0].pbrMetallicRoughness.baseColorTexture.setTexture(tex);
  } catch(e){}
}

/* ── CANVAS BG ── */
function setCanvasBg(hex) {
  S.canvasBg = hex;
  fC.setBackgroundColor(hex, () => { fC.renderAll(); sync(); });
}

/* ── TEXT ── */
async function addText() {
  await document.fonts.ready;
  const txt = document.getElementById('ti').value || 'Your Text';
  const color = document.getElementById('tc').value;
  const font = document.getElementById('tf').value;
  const size = parseInt(document.getElementById('ts').value);
  const o = new fabric.Text(txt, {
    left:512, top:512, originX:'center', originY:'center',
    fontFamily:font, fontSize:size, fill:color, fontWeight:700
  });
  fC.add(o); fC.setActiveObject(o); fC.renderAll(); sync();
  showToast('Text added — go to Canvas tab to reposition');
}

/* ── LOGO ── */
function addLogo(inp) {
  const file = inp.files[0]; if (!file) return;
  const reader = new FileReader();
  reader.onload = e => {
    const src = e.target.result;
    document.getElementById('lpi').src = src;
    document.getElementById('lpw').style.display = 'block';
    fabric.Image.fromURL(src, img => {
      if (img.width > 400) img.scaleToWidth(400);
      img.set({ left:512, top:512, originX:'center', originY:'center' });
      fC.add(img); fC.setActiveObject(img); logoObj = img;
      fC.renderAll(); sync();
      showToast('Logo added — drag on canvas to reposition');
    });
  };
  reader.readAsDataURL(file);
}

function scaleLogo(v) {
  const o = logoObj || fC.getActiveObject();
  if (o) { o.scale(parseFloat(v)); fC.renderAll(); sync(); }
}

/* ── TOGGLES ── */
function doTog(k) {
  S[k] = !S[k];
  document.getElementById('tog-'+k).classList.toggle('on', S[k]);
  if (k === 'rotate') {
    if (S.rotate) mv.setAttribute('auto-rotate','');
    else mv.removeAttribute('auto-rotate');
  }
}

/* ── SAVE ── */
async function saveDesign() {
  // snapshot at lower res for storage
  const snap = fC.toDataURL({ format:'png', quality:0.75, multiplier:0.4 });

  // Collect mat colors
  const matColors = Array.from(document.querySelectorAll('.mi input[type=color]')).map(i => i.value);

  const d = {
    id: 'D' + Date.now(),
    ts: new Date().toISOString(),
    size: S.size,
    qty,
    primaryColor: S.primaryColor,
    secondaryColor: S.secondaryColor,
    canvasBg: S.canvasBg,
    matColors,
    cj: fC.toJSON(),
    snap,
  };

  designs.push(d);
  try { localStorage.setItem('gd', JSON.stringify(designs)); } catch(e) {
    // If storage is full, remove oldest
    designs.shift();
    designs.push(d);
    localStorage.setItem('gd', JSON.stringify(designs));
  }
  updBadge();
  showToast('✓ Design #' + designs.length + ' saved!');
}

  a.download = `golf-design-${Date.now()}.png`;
  a.click();
  showToast('📥 Design canvas exported');
}

/* ── RENDER SAVED ── */
function renderSaved() {
  const grid = document.getElementById('sgr');
  const es = document.getElementById('es');
  document.getElementById('sc-ct').textContent = designs.length + ' design' + (designs.length!==1?'s':'');

  if (!designs.length) {
    grid.innerHTML = '';
    grid.appendChild(es);
    es.style.display = 'flex';
    return;
  }
  es.style.display = 'none';
  grid.innerHTML = '';

  designs.forEach((d, i) => {
    const ts = new Date(d.ts);
    const date = ts.toLocaleDateString('en-US',{month:'short',day:'numeric',year:'numeric'});
    const time = ts.toLocaleTimeString('en-US',{hour:'2-digit',minute:'2-digit'});
    const swatches = (d.matColors||[d.primaryColor,d.secondaryColor]).map(c=>`<div class="dcs-sw" style="background:${c};"></div>`).join('');

    const card = document.createElement('div');
    card.className = 'sdc';
    card.innerHTML = `
      <div class="sdcp">
        ${d.snap ? `<img src="${d.snap}" alt="Design ${i+1}">` : `<div style="color:var(--mu);font-size:11px;">No preview</div>`}
        <div style="position:absolute;top:7px;right:7px;background:rgba(0,0,0,.7);color:var(--ac);font-size:9px;font-weight:600;padding:2px 7px;border-radius:8px;letter-spacing:.05em;">
          #${i+1}
        </div>
      </div>
      <div class="sdcb">
        <div class="ds">Design #${i+1}</div>
        <div class="dd">${date} · ${time}</div>
        <div class="dcs">${swatches}</div>
        <div class="di">Size: ${d.size||'M'} &nbsp;·&nbsp; Qty: ${d.qty||1}</div>
        <div class="da">
          <button class="da-btn da-e" onclick="loadDesign(${i})">✎ Edit</button>
          <button class="da-btn da-d" onclick="delDesign(${i})">✕ Delete</button>
        </div>
      </div>`;
    grid.appendChild(card);
  });
}

function delDesign(i) {
  designs.splice(i,1);
  localStorage.setItem('gd', JSON.stringify(designs));
  renderSaved(); updBadge();
  showToast('Design deleted');
}

function loadDesign(i) {
  const d = designs[i];
  if (!d) return;
  showPg('c', document.querySelector('.ntab'));

  fC.loadFromJSON(d.cj, () => { fC.renderAll(); sync(); });

  if (d.matColors) {
    d.matColors.forEach((c, mi) => {
      const mat = mats[mi];
      if (!mat) return;
      if (mi === 0) { setCanvasBg(c); document.getElementById('cbg').value = c; }
      else mat.pbrMetallicRoughness.setBaseColorFactor(c);
      const cp = document.querySelector(`.mi[data-i="${mi}"] input[type=color]`);
      if (cp) cp.value = c;
    });
  }

  if (d.size) {
    const btn = Array.from(document.querySelectorAll('.spi')).find(b => b.textContent.trim() === d.size);
    if (btn) { btn.click(); }
  }

  qty = d.qty || 1;
  document.getElementById('qty-n').textContent = qty;
  showToast('Design #' + (i+1) + ' loaded for editing');
}

function updBadge() {
  const b = document.getElementById('sbadge');
  if (designs.length > 0) { b.style.display='inline'; b.textContent=designs.length; }
  else b.style.display='none';
}

/* ── NAVIGATION ── */
function showPg(name, btn) {
  document.querySelectorAll('.ntab').forEach(t => t.classList.remove('on'));
  document.querySelectorAll('.pg').forEach(p => p.classList.remove('on'));
  document.getElementById('pg-'+name).classList.add('on');
  // activate right tab button
  document.querySelectorAll('.ntab').forEach(t => {
    if (t.getAttribute('onclick') && t.getAttribute('onclick').includes("'"+name+"'")) t.classList.add('on');
  });
  if (name === 's') renderSaved();
}

function swDT(name, btn) {
  document.querySelectorAll('.dta').forEach(t => t.classList.remove('on'));
  document.querySelectorAll('.dtc').forEach(c => c.classList.remove('on'));
  btn.classList.add('on');
  document.getElementById('dtc-'+name).classList.add('on');
  if (name === 'canvas') {
    setTimeout(() => {
      const el = document.getElementById('co');
      if (!el || !el.clientWidth) return;
      const w = el.clientWidth;
      const z = w/1024;
      fC.setZoom(z);
      fC.setWidth(w);
      fC.setHeight(w);
    }, 60);
  }
}

/* ── TOAST ── */
function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('on');
  clearTimeout(window._tt);
  window._tt = setTimeout(() => t.classList.remove('on'), 3200);
}
